old = """    const igData = await fetchIG('topseller=1')
    const result = igData
      .filter(g => {
        if (g.topseller !== 1 || g.stock !== 1 || parseFloat(g.price) <= 0) return false
        const region = (g.region || '').toLowerCase()
        if (region.includes('latin')) return false
        if (region.includes('us') && !region.includes('europe') && !region.includes('australia')) return false
        return true
      })
      .sort((a, b) => b.id - a.id)
      .slice(0, 50)
    setCache('topsellers', result)"""

new = """    const igData = await fetchIG('topseller=1')
    const games = igData
      .filter(g => {
        if (g.topseller !== 1 || g.stock !== 1 || parseFloat(g.price) <= 0) return false
        return !isExcluded(g)
      })
      .slice(0, 100)
    const withDates = await enrichGames(games)
    const today = new Date()
    const currentYear = today.getFullYear()
    const result = withDates
      .filter(game => {
        const year = getYear(game.releaseDate)
        if (year !== currentYear) return false
        return new Date(game.releaseDate) <= today
      })
      .sort((a, b) => new Date(b.releaseDate) - new Date(a.releaseDate))
    setCache('topsellers', result)"""
